import java.util.Scanner;

public class Main
{
    public static void temperaturas() {
        float[][] temperatura = new float[10][3];
        Scanner teclado = new Scanner(System.in);
        
        int opcao = 0;
        
        for(var x=0; x<10; x++){
            System.out.println("Informe a temperatura minima do dia "+ (x+1)+":");
            temperatura[x][0] = teclado.nextFloat();
            
             System.out.println("Informe a temperatura máxima do dia: "+ (x+1)+":");
            temperatura[x][1] = teclado.nextFloat();
            
            temperatura[x][2] = (temperatura[x][0]+temperatura[x][1])/2;
        }
        
        do {

            System.out.println("\n----- MENU -----");
            System.out.println("1 - Temperatura minima de todos os dias");
            System.out.println("2 - Temperatura maxima de todos os dias");
            System.out.println("3 - Maior variacao de temperatura");
            System.out.println("4 - Sair do programa");
            System.out.println("Escolha uma opcao:");

            opcao = teclado.nextInt();

            switch (opcao) {
                case 1:
                    for (var x = 0; x < 10; x++) {
                        System.out.println("Dia " + (x+1) + ": " + temperatura[x][0]);
                    }
                    break;

                case 2:
                    for (var x = 0; x < 10; x++) {
                        System.out.println("Dia " + (x+1) + ": " + temperatura[x][1]);
                    }
                    break;
                    
                 case 3:
                    float maiorVariacao = temperatura[0][1] - temperatura[0][0];
                    int dia = 1;

                    for (var x = 1; x < 10; x++) {

                        float variacao = temperatura[x][1] - temperatura[x][0];

                        if (variacao > maiorVariacao) {
                            maiorVariacao = variacao;
                            dia = x + 1;
                        }
                    }
                    System.out.println("Maior variacao: " + maiorVariacao);
                    System.out.println("Dia: " + dia);
                    break;
                    
                case 4:
                    System.out.println("Programa encerrado!");
                    break;

                default:
                    System.out.println("Opcao invalida!");
                    
            }
            
        } while (opcao != 4);
        teclado.close();
    }
    
    public static void calculaTriangulo() {

        float[][] triangulo = new float[10][3];
        Scanner teclado = new Scanner(System.in);

        int equilatero = 0;
        int isosceles = 0;
        int escaleno = 0;
        int invalido = 0;

        // Cadastro das arestas
        for (var x = 0; x < 10; x++) {

            System.out.println("Informe a primeira aresta do triangulo " + (x+1) + ":");
            triangulo[x][0] = teclado.nextFloat();

            System.out.println("Informe a segunda aresta do triangulo " + (x+1) + ":");
            triangulo[x][1] = teclado.nextFloat();

            System.out.println("Informe a terceira aresta do triangulo " + (x+1) + ":");
            triangulo[x][2] = teclado.nextFloat();

            // Verificacao e classificacao
            if (triangulo[x][0] <= 0 || triangulo[x][1] <= 0 || triangulo[x][2] <= 0 ||
                triangulo[x][0] + triangulo[x][1] <= triangulo[x][2] ||
                triangulo[x][0] + triangulo[x][2] <= triangulo[x][1] ||
                triangulo[x][1] + triangulo[x][2] <= triangulo[x][0]) {

                invalido++;

            } else if (triangulo[x][0] == triangulo[x][1] &&
                       triangulo[x][1] == triangulo[x][2]) {

                equilatero++;

            } else if (triangulo[x][0] == triangulo[x][1] ||
                       triangulo[x][0] == triangulo[x][2] ||
                       triangulo[x][1] == triangulo[x][2]) {

                isosceles++;

            } else {

                escaleno++;
            }
        }

        // Calculo dos percentuais
        System.out.println("\n----- RESULTADO -----");

        System.out.println("Percentual de triangulos equilateros: " + (equilatero * 10) + "%");

        System.out.println("Percentual de triangulos isosceles: " + (isosceles * 10) + "%");

        System.out.println("Percentual de triangulos escalenos: " + (escaleno * 10) + "%");

        System.out.println("Triangulos invalidos: " + invalido);

        teclado.close();
    }

    
    
    
	public static void main(String[] args) {
		temperaturas();
		calculaTriangulo();
	}
}
